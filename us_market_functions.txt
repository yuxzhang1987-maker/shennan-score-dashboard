US_MARKET_ITEMS = {
    "费城半导体指数SOX评分": ("^SOX", 0.40),
    "纳斯达克100指数评分": ("^NDX", 0.25),
    "英伟达NVDA评分": ("NVDA", 0.15),
    "AMD评分": ("AMD", 0.10),
    "博通AVGO评分": ("AVGO", 0.10),
}

def get_yahoo_daily(symbol):
    """通过Yahoo公开Chart接口获取美股日线；返回收盘价序列。"""
    url = f"https://query1.finance.yahoo.com/v8/finance/chart/{symbol}"
    payload = http_get(url, {"range": "1y", "interval": "1d", "events": "history"}, timeout=20).json()
    result = ((payload.get("chart") or {}).get("result") or [])
    if not result:
        err = (payload.get("chart") or {}).get("error")
        raise RuntimeError(f"Yahoo未返回{symbol}行情：{err}")
    item = result[0]
    timestamps = item.get("timestamp") or []
    quote = ((item.get("indicators") or {}).get("quote") or [{}])[0]
    closes = quote.get("close") or []
    dates = pd.to_datetime(timestamps, unit="s", utc=True).tz_convert("America/New_York").tz_localize(None).normalize()
    frame = pd.DataFrame({"日期": dates, "收盘": closes})
    frame["收盘"] = as_num(frame["收盘"])
    series = frame.dropna(subset=["日期", "收盘"]).drop_duplicates("日期").set_index("日期")["收盘"].sort_index()
    if len(series) < 55:
        raise RuntimeError(f"{symbol}有效历史不足55个交易日")
    return series

def get_us_tech():
    """获取SOX、NDX及三只美国半导体/AI龙头股；单项失败时保留其余有效数据。"""
    data, sources, errors = {}, {}, []
    for label, (symbol, weight) in US_MARKET_ITEMS.items():
        try:
            data[label] = get_yahoo_daily(symbol)
            sources[label] = f"Yahoo Finance {symbol}"
            continue
        except Exception as yahoo_error:
            if symbol in ("^SOX", "^NDX") and ak is not None:
                try:
                    sina_symbol = ".SOX" if symbol == "^SOX" else ".NDX"
                    df = safe_ak_call(ak.index_us_stock_sina, symbol=sina_symbol)
                    date_col = "date" if "date" in df.columns else df.columns[0]
                    df = df.copy()
                    df[date_col] = pd.to_datetime(df[date_col], errors="coerce").dt.normalize()
                    df["close"] = as_num(df["close"])
                    series = df.dropna(subset=[date_col, "close"]).sort_values(date_col).set_index(date_col)["close"]
                    if len(series) < 55:
                        raise RuntimeError("备用指数历史不足55日")
                    data[label] = series
                    sources[label] = f"新浪财经AKShare {sina_symbol}"
                    continue
                except Exception as fallback_error:
                    errors.append(f"{label}：Yahoo失败({yahoo_error})；新浪备用失败({fallback_error})")
            else:
                errors.append(f"{label}：{yahoo_error}")
    return data, sources, errors

def trend_score(series):
    """美股市场趋势评分：20日收益与50日均线位置。"""
    s = as_num(series).dropna().sort_index()
    ret20 = s.pct_change(20) * 100
    ma50 = s.rolling(50, min_periods=25).mean()
    gap50 = (s / ma50 - 1) * 100
    return 0.60 * _score_pct(ret20, -10, 10) + 0.40 * _score_gap(gap50, -5, 5)
